// ИСПРАВЛЕННЫЙ МОДУЛЬ АУДИОПЛЕЕРА let currentAudio = null; let playSessionId = 0; // Идентификатор сессии для отсечения фоновых призраков function stopAudio() { playSessionId++; // Инвалидируем все предыдущие асинхронные попытки if (currentAudio) { currentAudio.pause(); currentAudio.oncanplay = null; currentAudio.onended = null; currentAudio.onerror = null; currentAudio.src = ""; // Физически выгружаем поток из памяти currentAudio.load(); currentAudio = null; } voiceBtn.classList.remove("speaking"); voiceBtn.textContent = "Воспроизвести маршрут"; } voiceBtn.addEventListener("click", () => { // 1. Если трек уже играет — ставим на паузу if (currentAudio && !currentAudio.paused) { currentAudio.pause(); voiceBtn.classList.remove("speaking"); voiceBtn.textContent = "Продолжить"; return; } // 2. Если трек на паузе — снимаем с паузы if (currentAudio && currentAudio.paused) { currentAudio.play().then(() => { voiceBtn.classList.add("speaking"); voiceBtn.textContent = "Пауза"; }).catch(() => stopAudio()); return; } // 3. Запуск нового воспроизведения const fromKey = fromEl.value; const toKey = toEl.value; if (fromKey === toKey) return; const base = `${fromKey}_${toKey}`; const candidates = [ `audio/${base}.mp3.mp3`, `audio/${base}.mp3`, `${base}.mp3.mp3`, `${base}.mp3` ]; stopAudio(); // Полный сброс перед стартом const thisSession = playSessionId; tryPlayAudioFile(candidates, 0, thisSession); }); function tryPlayAudioFile(candidates, idx, sessionId) { // Если сессия сменилась (пользователь нажал стоп/сменил маршрут) — глушим процесс if (sessionId !== playSessionId) return; if (idx >= candidates.length) { toastMsg("Аудиозапись для этого маршрута пока не добавлена."); stopAudio(); return; } const audio = new Audio(); currentAudio = audio; let hasStarted = false; // Защита от циклического перезапуска по событию буферизации audio.oncanplay = () => { if (sessionId !== playSessionId) { audio.src = ""; return; } if (!hasStarted) { hasStarted = true; audio.play().then(() => { voiceBtn.classList.add("speaking"); voiceBtn.textContent = "Пауза"; }).catch((err) => { console.warn("Автовоспроизведение заблокировано браузером:", err); stopAudio(); }); } }; // Окончание трека: очищаем состояние строго один раз audio.onended = () => { if (sessionId === playSessionId) { stopAudio(); } }; // Ошибка загрузки конкретного пути: пробуем следующий кандидат audio.onerror = () => { if (sessionId === playSessionId) { audio.src = ""; tryPlayAudioFile(candidates, idx + 1, sessionId); } }; audio.preload = "auto"; audio.src = candidates[idx]; }
